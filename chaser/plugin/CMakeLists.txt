cmake_minimum_required(VERSION 3.15)
project(VJChaser LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(FFGL ${CMAKE_CURRENT_SOURCE_DIR}/ffgl)
file(GLOB FFGL_SRC ${FFGL}/source/lib/ffgl/*.cpp ${FFGL}/source/lib/ffglex/*.cpp ${FFGL}/source/lib/ffglquickstart/*.cpp)
add_library(VJChaser SHARED VJChaser.cpp ${FFGL_SRC} ${FFGL}/build/windows/FFGLPlugins.def)
target_include_directories(VJChaser PRIVATE ${FFGL}/source/lib ${FFGL}/deps/glew-2.1.0/include)
target_compile_definitions(VJChaser PRIVATE GLEW_STATIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_link_libraries(VJChaser PRIVATE ${FFGL}/deps/glew-2.1.0/lib/Release/x64/glew32s.lib opengl32)
set_property(TARGET VJChaser PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
